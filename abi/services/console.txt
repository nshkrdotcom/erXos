# The console interface (docs/design-live.md, "The tty"): the six messages a
# tty and its master exchange, each heard through the other's domain envelope and
# checked by its receiver as that section's table says. A tty sends tty_open and
# tty_output; its master sends tty_taken, tty_input, tty_size and tty_hangup.
service console 11
  served_as message
  operation tty_open 1
  operation tty_output 2
  operation tty_taken 3
  operation tty_input 4
  operation tty_size 5
  operation tty_hangup 6
# A tty's window size is Rows then Columns, the order of POSIX's struct winsize, in
# tty_size and wherever erXos holds a size; each side is 1 to console_side_max
# (docs/design-live.md, "Between domains"). Only OTP's own code takes {Columns, Rows}:
# prim_tty, a group's tty_geometry, and ssh_connection's pty and window_change.
limit console_side_max u32 65535
