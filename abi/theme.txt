# The desktop's theme (docs/design-desktop-programs.md, "The theme: one
# declaration", and docs/design-abi.md, "Theme"): the type scale, the
# spacing grid, the radii, the shadows, the one light palette with the
# contrast it must meet, and the fonts and icons the glyph step
# rasterises. tools/gen_abi.rb validates every row and generates
# erx_theme in erx_memdraw, erx/theme.h and the web client's theme.ts.

theme 1
  # Pixels per spacing unit; a role's line height lies on this grid.
  unit 4
  # A separator's width, in pixels.
  hairline 1
  # A font: its name, its directory, and the file of each weight it has.
  font ui display/fonts/inter regular Inter-Regular.ttf medium Inter-Medium.ttf semibold Inter-SemiBold.ttf
  # The code points the glyph step rasterises from a font.
  range ui 0020-007E 00A0-017F 2000-206F 2190-21FF
  # Type roles; a renderer never picks a size, a role does. Sizes and line
  # heights are pixels; a role without a colour takes `text`.
  role display font ui size 48 line 64 weight semibold figures tabular
  role title   font ui size 20 line 28 weight semibold
  role heading font ui size 15 line 20 weight semibold
  role body    font ui size 13 line 20 weight regular
  role numeric font ui size 13 line 20 weight medium figures tabular
  role caption font ui size 11 line 16 weight regular color secondary
  # The spacing scale, in units.
  space xs 1 s 2 m 3 l 4 xl 6 xxl 8
  # Corner radii, in pixels; windows are square.
  radius control 6 section 10 menu 8 window 0
  # Shadows: x and y offset and blur in pixels, and the opacity of black in percent.
  shadow section 0 2 12 10
  # The palette, as sRGB RRGGBB.
  color window     F6F5F2
  color surface    FFFFFF
  color sunken     EEEDE9
  color line       DDDCD7
  color text       1B1C1E
  color secondary  5D6066
  color tertiary   8A8E94
  color accent     0E7A6B
  color on_accent  FFFFFF
  color ok         2F7D4F
  color warn       9A6A12
  color critical   B3362C
  color series_cpu 0E7A6B
  color series_mem 5B53C6
  color series_msg B8620B
  color desktop    D9DAD5
  # The WCAG 2 contrast ratio each foreground meets on each background named.
  contrast text      on window surface sunken 7.0
  contrast secondary on window surface sunken 4.5
  contrast on_accent on accent 4.5
  # The icons' definition file and the pixel sizes they are rasterised at.
  icons display/icons/icons.txt sizes 16 32
