# A start row's restart intensity, more than restarts ends within period seconds
# (docs/design-apps.md, "The start table"), from the manifest or from rc.conf. Its
# row supervisor keeps a timestamp per end within the period, so the ceilings bound
# that list and the restart rate of a row whose start is refused at once: at most
# 100 ends, enough for any session that is worth keeping, within at most a day,
# past which ends are not the same burst. The reader, the packer and the start
# table each refuse a value past them.
limit start_restarts_max u32 100
limit start_period_max_s u32 86400
# The intensity of a start row that states none, a manifest row's or a package's
# (docs/design-packaging.md, "The declaration"): 5 ends within 600 seconds.

limit start_period_default_s u32 600
# The seconds a domain's init has to report ready to the domain service: the
# machine's boot_timeout_s, which every configured domain, session and create
# request that sets none takes (docs/design-system.md, "The manifest"). The
# manifest source may leave it out for the default, and the reader bounds it to
# 1 .. boot_timeout_max_s, so its milliseconds fit an Erlang timer. The default
# suits hardware and QEMU: virt under TCG on a loaded host takes longer than
# 10 s to boot a BeamAsm domain, and a longer one only notices a hung boot
# later. A machine much slower than those, as the website's emulator is, states
# its own.
limit boot_timeout_default_s u32 60
limit boot_timeout_max_s u32 3600

# A partition window's role (docs/design-abi.md, "Devices and partition windows"): the
# slots' boot and system partitions and the two records are the slot writer's, which
# only the system domain holds in Phase 1 (docs/design-storage.md, PW1): a configured
# domain holds only a configured role. A window of a role not marked writable grants
# read alone, and its block driver refuses every write to it, so a role added with
# neither attribute is the slot writer's and read-only. system is the one window on the
# system partition of the slot the boot value names, and an unknown boot has none.
# verified_image is a window on one EROFS image of the booted slot
# (docs/todo-system-partition.md, deletions). record_volume is the one window on the
# whole record partition besides the slot writer's two, read-only, held by the domain
# that holds the system window, which reads root's login keys through it
# (docs/design-ssh.md, "Root's login keys on the medium").
enum mf_window_role u32
  attribute configured
  attribute writable
  value boot_a 1 writable
  value boot_b 2 writable
  value boot_selection 3 writable
  value reset_count 4 writable
  value vault 5 configured writable
  value data 6 configured writable
  value fat 7 configured writable
  value system 8 configured
  value system_a 9 writable
  value system_b 10 writable
  value verified_image 11 configured
  value record_volume 12 configured
