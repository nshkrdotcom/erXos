# SSH's constants: the key blob and the key count that the readers of root's login keys
# share. A value erXos has no use for is not entered.
#
# The source is RFC 8709 (rfc8709.txt, SHA-256
# ea4aa3a8feec6bd5b1cc7847a6d08cd0afae49d12220cac1e2c39bc72c29fd3d), from
# https://www.rfc-editor.org/rfc/, sections cited. A value marked erXos's is the
# design's own choice, not the specification's.

# The name of the one login key algorithm, which opens the key blob (RFC 8709,
# section 4), and the size of the key that blob carries after it.
bytes ssh_ed25519_name 11 value 7373682d65643235353139
limit ssh_ed25519_key_size u32 32

# erXos's: the most keys root's authorized_keys names, and so the most key blobs the
# login_keys tree serves and a reader of that tree takes (docs/design-ssh.md, "Root's
# login keys on the medium").
limit ssh_authorized_keys_max u32 16
