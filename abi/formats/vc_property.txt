# The Raspberry Pi firmware's mailbox property buffer (docs/design-board-sensors.md, "The
# requests" and "The kernel call"; docs/design-kernel.md, "Firmware requests"), as the
# mailbox core of erx_firmware_request writes a request and checks its answer.
#
# The buffer's facts are the firmware wiki's page "Mailbox property interface"
# (https://raw.githubusercontent.com/wiki/raspberrypi/firmware/Mailbox-property-interface.md,
# fetched 2026-11-14, SHA-256
# 1197d7f52ae2ce6b731bb1ea8e1fce93df4395983629994069e71d498160dd8b), at the line cited, as
# docs/design-board-sensors.md, "What the firmware offers", reads it. Every word is
# little-endian, and the buffer is 16-byte aligned (line 3).

# The buffer's header (lines 22-34): its size in bytes, the whole buffer from this word to
# the end tag, and the request or response code.
layout erx_vc_header wire_le 4 8
  field size u32 0
  field code u32 4

# A request's code, and the response code of a buffer the firmware answered (lines 29-30).
# Any other response code, its error parsing the buffer 0x80000001 among them, is an
# answer outside the interface.
limit vc_code_request u32 0
limit vc_code_success u32 0x80000000

# One tag's header (lines 36-47): its identifier, the size of its value buffer in bytes,
# and the request or response word, 0 in a request. In a response, bit 31 is set and bits
# 30-0 give the value's length; a tag the firmware ignores keeps bit 31 clear (line 5). A
# length above the value buffer's size is the truncation the interface defines (lines
# 7-10). The value buffer follows, a whole number of words.
layout erx_vc_tag_header wire_le 4 12
  field tag u32 0
  field value_size u32 4
  field indicator u32 8
    bits length 0 30
    bits response 31 31

# The end tag closes the tag sequence (lines 22-34).
limit vc_end_tag u32 0

# The largest answer window a tag declares, erXos's choice (docs/design-kernel.md,
# "Firmware requests"), and the largest request, which is the gencmd tag's 64 bytes
# (docs/design-board-sensors.md, "The requests").
limit vc_window_max u32 64
limit vc_request_max u32 64

# How long one request may take, from the call's start to its answer, the wait for room
# to post included (docs/design-kernel.md, "Firmware requests"): erXos's bound for every
# mailbox request, in nanoseconds.
limit vc_deadline_ns u64 1000000000

# The tags erXos sends, a closed set, each member's value its identifier: a firmware request
# row of the system manifest names one (abi/manifest.txt, erx_mf_firmware_request), and no
# member reads the board's key. Each tag declares the bytes of its request, of its answer
# window and of its least answer, and its result_word: 1 when the firmware answers the tag
# in its first value word rather than in its response word, 0 otherwise. The words of a row
# are the platform record's or a power line's below, never the caller's.
#
# NOTIFY_XHCI_RESET has the USB controller behind a PCIe bridge load its firmware: its
# request is one word, the function's address bus << 20 | device << 15 | function << 12,
# and its answer is empty (docs/research-brcmstb-pcie.md, section 5; docs/design-usb.md,
# "The VL805's firmware").
#
# SET_GPIO_STATE sets a pin of the firmware's GPIO expander: its request is two words, the
# pin and the state, and its answer is its first value word, which the firmware writes 0
# when it performed the set; the firmware does not set the response word (FreeBSD
# releng/15.1, sys/arm/broadcom/bcm2835/: bcm2835_firmware.h, lines 161-168;
# bcm2835_mbox.c, lines 362-370; raspberrypi_gpio.c, lines 253-255; probe B-P2 of
# docs/research-bluetooth-pi4.md, "The power phases"). A buffer the firmware left untouched
# keeps the pin the request wrote, so an ignored request is not taken as performed. Its
# rows come only from a power line (fw_power_line below), never from a record row
# (docs/design-bluetooth.md, "The power line").
enum fw_tag u32
  value notify_xhci_reset 0x00030058
  value set_gpio_state 0x00038041
limit vc_notify_xhci_reset_request u32 4
limit vc_notify_xhci_reset_window u32 0
limit vc_notify_xhci_reset_least u32 0
limit vc_notify_xhci_reset_result_word u32 0
limit vc_set_gpio_state_request u32 8
limit vc_set_gpio_state_window u32 0
limit vc_set_gpio_state_least u32 0
limit vc_set_gpio_state_result_word u32 1

# The most request words a device row carries (abi/manifest.txt, erx_mf_firmware_request,
# argument0 and argument1); the manifest generator refuses a device row whose tag's
# request is longer.
limit vc_row_words u32 2

# The power lines a platform record's power row may name, a closed set: each a pin of the
# firmware's GPIO expander, set with SET_GPIO_STATE, with the pin's request word and its
# off and on states. The manifest generator derives a device's rows from its power row,
# one per phase of fw_power_phase below at the phase's index, named after the line
# (docs/design-bluetooth.md, "The power line"). bt_power is the Raspberry Pi 4's Bluetooth controller's: expander pin
# 0, request pin 128, which the firmware's device tree names in the shutdown-gpios of
# UART0's bluetooth child, gpio-line-names BT_ON, active high (probe B-P2 of
# docs/research-bluetooth-pi4.md, "The power phases"); nothing reads the tree for it. A
# line's pin word is never 0, which the generator refuses: an untouched buffer keeps the
# pin, and a 0 there would read as performed.
enum fw_power_line u32
  value bt_power 0
limit vc_bt_power_pin u32 128
limit vc_bt_power_off u32 0
limit vc_bt_power_on u32 1

# A power line's phases, a closed set: each phase's value is the index of its row among
# the device's rows, which the manifest generator writes in value order, and the device
# service asks for a phase by that index (erx_domain_lib:firmware/2); the domain library
# accepts a phase only for a device whose manifest row carries a power line's times
# (docs/design-bluetooth.md, "The power line", "Authority").
enum fw_power_phase u32
  value off 0
  value on 1
