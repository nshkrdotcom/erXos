# The xHCI memory structures the device NIF builds and reads (docs/design-usb.md, "The
# TRB ring form"): the slot, endpoint and input control contexts, the event ring's
# segment table entry, the TRBs by shape, and the values their fields take.
#
# Each field and value is the one FreeBSD gives, at releng/15.1 (commit 95719d8a895b),
# in sys/dev/usb/controller/xhci.h and xhci.c at the lines cited. Where FreeBSD names
# no field, it is from the xHCI specification, revision 1.2 (May 2019), by section
# and table. Every structure is little-endian. A controller whose HCCPARAMS1.CSZ is
# set spaces its contexts 64 bytes apart, each context's first 32 bytes the layouts
# below and the rest reserved (xhci.h:113-116, :163-166, :182-185). The ring form's own
# sizes, its rings' lengths and queue blocks, are its record's, not this file's
# (design-usb.md, "The TRB ring form"); the size of an interrupt and of a bulk queue
# block's buffer is here, since the USB core admits an endpoint only when its payload of one service
# interval fits one.

# A device context holds the slot context and 31 endpoint contexts, indexed by the
# device context index, EP0's 1 (XHCI_MAX_ENDPOINTS, xhci.h:33); an input context
# holds the input control context before them.
limit xhci_device_contexts u32 32

# A route string names a hub port in four bits for each tier below the root, the first
# tier lowest (xhci.c:2636-2644).
limit xhci_route_tier_bits u32 4

# Alignment and boundaries (xhci.h:48-61; xHCI 1.2, Table 6-1): no ring segment
# spans a 64 KiB boundary, and no context, context array or DCBAA a page.
limit xhci_page_size u32 4096
limit xhci_segment_boundary u32 65536
limit xhci_dcbaa_align u32 64
limit xhci_device_context_align u32 64
limit xhci_input_context_align u32 64
limit xhci_transfer_ring_align u32 16
limit xhci_command_ring_align u32 64
limit xhci_event_ring_align u32 64
limit xhci_segment_table_align u32 64
limit xhci_scratchpad_array_align u32 64

# The error count every endpoint context the NIF builds takes (xhci.c:2495-2501).
limit xhci_endpoint_error_count u32 3

# A command the controller has not answered within this many milliseconds fails the
# controller, whose restart resets it (design-usb.md, "The controller process"), an erXos
# choice.
limit xhci_command_deadline_ms u32 5000

# The controller's start (design-usb.md, "The controller process"): it polls the BIOS's
# semaphore and calls the install again every 10 ms, FreeBSD's pause in both loops
# (xhci_pci.c:529-557, xhci.c:430-431); it goes on without the BIOS's release after 5 s,
# FreeBSD's 500 polls; and the install, three waits of FreeBSD's 1 s each (xhci.c:430-436,
# :469-477, :494-499) that one {error, ebusy} does not tell apart, fails the controller
# when it has not answered ok within their sum.
limit xhci_poll_ms u32 10
limit xhci_bios_handoff_ms u32 5000
limit xhci_install_deadline_ms u32 3000

# A running controller's USBSTS is read on every interrupt and, whether interrupts come or
# not, this often
# (design-usb.md, "The controller process"), so a controller that stopped without an
# interrupt, on a host controller error or a halt, or that left the bus, fails within
# about a second; an erXos choice. Events are never polled.
limit xhci_status_interval_ms u32 1000

# An interrupt pass stops and acknowledges the line once it has read this many events: two
# wraps of the event ring's 512 entries (design-usb.md, "The TRB ring form"), where
# FreeBSD's xhci_interrupt_poll stops (xhci.c:1138, :1167-1176).
limit xhci_events_per_interrupt u32 1024

# The buffer of an interrupt and of a bulk queue block: an interrupt block's four of
# 1 KiB and a bulk block's two of 32 KiB (design-usb.md, "The TRB ring form"), each an
# erXos choice.
limit xhci_interrupt_buffer_size u32 1024
limit xhci_bulk_buffer_size u32 32768
# A control block's one buffer of 4 KiB, which bounds a control transfer's data
# stage (design-usb.md, "The TRB ring form"), an erXos choice.
limit xhci_control_buffer_size u32 4096

# An interrupt endpoint's Interval, the base-2 exponent of its period in 125 us units
# (xHCI 1.2, 6.2.3.6 and Table 6-12, p. 456). At low and full speed bInterval counts
# milliseconds, 1 to 255, and the exponent is the base-2 logarithm of bInterval times 8
# (8 being 2 to the frame shift), rounded down, so 3 to 10; at high and SuperSpeed
# bInterval is 1 to its maximum and the exponent is bInterval less one, 0 to 15. A bInterval
# outside its range is one the table gives no period for. A bulk endpoint's Interval is
# 0: the controller does not use it at SuperSpeed (6.2.3.6), and FreeBSD sets it only
# for periodic endpoints (xhci.c:2463-2470).
limit xhci_interval_frame_shift u8 3
limit xhci_interval_binterval_max u8 16

# The slot context (xhci.h:74-111): the device's route, speed and hub fields, its
# last valid context index, the root hub port it is reached through, its
# transaction translator, its interrupter, and the address and state the
# controller writes, which software sets to 0 in an input context (xHCI 1.2, Table 6-7,
# p. 447; section 4.5.3, p. 98).
layout erx_xhci_slot_context wire_le 4 32
  field route u32 0
    bits route_string 0 19
    bits speed 20 23
    bits multi_tt 25 25
    bits hub 26 26
    bits context_entries 27 31
  field ports u32 4
    bits exit_latency 0 15
    bits root_port 16 23
    bits port_count 24 31
  field tt u32 8
    bits tt_slot 0 7
    bits tt_port 8 15
    bits tt_think_time 16 17
    bits interrupter 22 31
  field state u32 12
    bits address 0 7
    bits slot_state 27 31
  padding 16 16
  states route.route_string
  states route.speed
  states route.multi_tt
  states route.hub
  states route.context_entries
  states ports.exit_latency
  states ports.root_port
  states ports.port_count
  states tt.tt_slot
  states tt.tt_port
  states tt.tt_think_time
  fixed tt.interrupter 0
  fixed state.address 0
  fixed state.slot_state 0

# The endpoint context (xhci.h:118-161). The dequeue pointer is 16-byte aligned and
# shares its word with the dequeue cycle state, both derived from the endpoint's bound
# ring. The endpoint state is the controller's, 0 in an input context (xHCI 1.2, Table
# 6-8, p. 450). Mult, MaxPStreams, LSA and HID (xhci.h:130-135, 143-144) are fixed at 0, since
# MaxPStreams above 0 makes the dequeue pointer a stream context array's address (xHCI
# 1.2, 6.2.3).
layout erx_xhci_endpoint_context wire_le 8 32
  field info u32 0
    bits endpoint_state 0 2
    bits mult 8 9
    bits max_pstreams 10 14
    bits lsa 15 15
    bits interval 16 23
  field config u32 4
    bits error_count 1 2
    bits endpoint_type 3 5
    bits hid 7 7
    bits max_burst 8 15
    bits max_packet 16 31
  field dequeue u64 8
    bits dequeue_cycle 0 0
    bits dequeue_pointer 4 63 shift 4 device_address
  field payload u32 16
    bits average_trb_length 0 15
    bits max_esit_payload 16 31
  padding 20 12
  fixed info.endpoint_state 0
  fixed info.mult 0
  fixed info.max_pstreams 0
  fixed info.lsa 0
  states info.interval
  states config.error_count
  states config.endpoint_type
  fixed config.hid 0
  states config.max_burst
  states config.max_packet
  states payload.average_trb_length
  states payload.max_esit_payload
  derived dequeue.dequeue_pointer
  derived dequeue.dequeue_cycle

# The input control context (xhci.h:168-180): bit n of each flag word drops or adds
# the context of device context index n, bit 0 the slot's in the add word.
layout erx_xhci_input_control_context wire_le 4 32
  field drop u32 0
  field add u32 4
  padding 8 24
  derived drop
  derived add

# One entry of the event ring segment table (xhci.h:396-400): the segment's address,
# 64-byte aligned (XHCI_EVENT_RING_SEG_ALIGN, xhci.h:56), and its size, which counts the
# segment's TRBs in bits 15:0 (xHCI 1.2, 6.5, Table 6-95 of the body text).
layout erx_xhci_segment_table_entry wire_le 8 16
  field base u64 0
    bits address 6 63 shift 6 device_address
  field size u32 8
    bits trbs 0 15
  padding 12 4
  derived base.address
  derived size.trbs

# Every TRB is 16 bytes, 16-byte aligned (XHCI_TRB_ALIGN, xhci.h:59): a 64-bit
# parameter, a status word and a control word whose cycle bit and type are in the
# same place in every format (xhci.h:228-277). Each layout below is one format, with the
# fields the NIF writes or reads; every bit it does not name the NIF leaves clear.
limit xhci_trb_align u32 16

# An entry of the device context base address array (xhci.h:48, 64): entry 0 holds the
# scratchpad array's address and entry k slot k's device context, each 64-byte aligned
# (XHCI_SCRATCH_BUF_ARRAY_ALIGN, XHCI_DEV_CTX_ALIGN, xhci.h:57, 49). The sequencer
# writes every entry from the layout alone.
layout erx_xhci_dcbaa_entry wire_le 8 8
  field entry u64 0
    bits address 6 63 shift 6 device_address
  derived entry.address

# An entry of the scratchpad buffer array (xhci.h:68): a scratchpad page's address, 4 KiB
# aligned (XHCI_PAGE_SIZE, xhci.h:61), which the sequencer writes from the layout alone.
layout erx_xhci_scratchpad_entry wire_le 8 8
  field entry u64 0
    bits page 12 63 shift 12 device_address
  derived entry.page

# The structures TRBs lie in (docs/design-device-dma.md, "What the formats say"): a
# transfer ring and the command ring, both written by the CPU, and the event ring, which
# the controller writes. The link TRB lies in every ring the CPU writes and is no
# meaning: the sequencer writes it from the layout alone. A transfer ring needs a
# binding: it takes runs only while a binding command has bound it to an endpoint.
target xhci_transfer_ring cpu needs_binding
target xhci_command_ring cpu
target xhci_event_ring device

# A normal TRB (xhci.h:235-277; xHCI 1.2, 6.4.1.1): a buffer's device address and
# length, the TD size, and the flags.
layout erx_xhci_trb_normal wire_le 8 16
  field buffer u64 0 device_address
  field status u32 8
    bits length 0 16
    bits td_size 17 21
    bits interrupter 22 31
  field control u32 12
    bits cycle 0 0
    bits interrupt_on_short 2 2
    bits chain 4 4
    bits interrupt_on_completion 5 5
    bits immediate_data 6 6
    bits type 10 15
  meaning xhci_transfer_ring
  selects control.type xhci_trb_type.normal
  refers buffer data length status.length
  states status.td_size
  states control.interrupt_on_short
  states control.interrupt_on_completion
  fixed status.interrupter 0
  fixed control.chain 0
  fixed control.immediate_data 0
  activates control.cycle

# A control transfer's data stage: a normal TRB's fields and the stage's direction
# (xhci.h:267, xhci.c:1940-1960).
layout erx_xhci_trb_data wire_le 8 16
  field buffer u64 0 device_address
  field status u32 8
    bits length 0 16
    bits td_size 17 21
    bits interrupter 22 31
  field control u32 12
    bits cycle 0 0
    bits interrupt_on_short 2 2
    bits chain 4 4
    bits interrupt_on_completion 5 5
    bits immediate_data 6 6
    bits type 10 15
    bits direction_in 16 16
  meaning xhci_transfer_ring
  selects control.type xhci_trb_type.data_stage
  refers buffer data length status.length
  states status.td_size
  states control.interrupt_on_short
  states control.interrupt_on_completion
  states control.direction_in
  fixed status.interrupter 0
  fixed control.chain 0
  fixed control.immediate_data 0
  activates control.cycle

# A control transfer's status stage: no buffer and no length, only its direction
# (xHCI 1.2, 6.4.1.2.3).
layout erx_xhci_trb_status wire_le 8 16
  padding 0 8
  field status u32 8
    bits interrupter 22 31
  field control u32 12
    bits cycle 0 0
    bits chain 4 4
    bits interrupt_on_completion 5 5
    bits type 10 15
    bits direction_in 16 16
  meaning xhci_transfer_ring
  selects control.type xhci_trb_type.status_stage
  states control.interrupt_on_completion
  states control.direction_in
  fixed status.interrupter 0
  fixed control.chain 0
  activates control.cycle

# A control transfer's setup stage: the 8 setup bytes as immediate data, little-endian
# in the parameter, and the transfer type of its data stage (xhci.h:230-277,
# xhci.c:1850-1870).
layout erx_xhci_trb_setup wire_le 8 16
  field setup u64 0
  field status u32 8
    bits length 0 16
    bits interrupter 22 31
  field control u32 12
    bits cycle 0 0
    bits interrupt_on_completion 5 5
    bits immediate_data 6 6
    bits type 10 15
    bits transfer_type 16 17
  meaning xhci_transfer_ring
  selects control.type xhci_trb_type.setup_stage
  states setup
  states control.interrupt_on_completion
  states control.transfer_type
  fixed status.length 8
  fixed status.interrupter 0
  fixed control.immediate_data 1
  activates control.cycle

# A link TRB: the segment it leads to, 16-byte aligned, and the toggle-cycle bit that
# flips the ring's cycle state when the controller follows it (xhci.h:241-277).
layout erx_xhci_trb_link wire_le 8 16
  field segment u64 0
    bits segment_pointer 4 63 shift 4 device_address
  field status u32 8
    bits interrupter 22 31
  field control u32 12
    bits cycle 0 0
    bits toggle_cycle 1 1
    bits chain 4 4
    bits interrupt_on_completion 5 5
    bits type 10 15
  selects control.type xhci_trb_type.link
  derived segment.segment_pointer
  fixed status.interrupter 0
  fixed control.toggle_cycle 1
  fixed control.chain 0
  fixed control.interrupt_on_completion 0
  activates control.cycle

# A command TRB (xhci.h:247-277): its parameter is the input context's address for
# Address Device, Configure Endpoint and Evaluate Context, and 0 for the others; its
# control word names the slot and, for Reset Endpoint and Stop Endpoint, the device
# context index. Enable Slot's bits 20:16 are its slot type, which the NIF writes 0. Bit
# 9, the block-set-address, deconfigure and transfer-state-preserve flag of the commands
# that have one, and bit 23, Stop Endpoint's suspend flag, stay clear.
layout erx_xhci_trb_command wire_le 8 16
  field parameter u64 0
    bits pointer 4 63 shift 4 device_address
  field status u32 8
  field control u32 12
    bits cycle 0 0
    bits type 10 15
    bits endpoint 16 20
    bits slot 24 31

# The commands (xHCI 1.2, 6.4.3), each a meaning of the command ring. A command's slot and
# endpoint are the sequencer's, written from its publication's reach, so the TRB names the
# endpoints the ledger closes and no other (docs/design-usb.md, "What the DMA check judges").
# Its reach mark says what its success or failure does to them: address binds EP0 a ring at
# the publication; configure binds a ring to each endpoint it adds and drops others; stop leaves its endpoint stopped;
# restart moves its endpoint's ring past its transfers; disable ends its slot. A command with
# no reach closes nothing. The flags a command may set, Address Device's block-set-address
# (Table 6-62), Configure Endpoint's deconfigure (Table 6-64), Reset Endpoint's
# transfer-state-preserve (Table 6-65) and Stop Endpoint's suspend (Table 6-66), are bits no
# layout names, so they stay clear. Every bit a table makes RsvdZ stays clear the same way.

# No Op (6.4.3.1, Table 6-58): the cycle bit and the type alone. It verifies the command ring
# (4.6.2). Figure 6-22 shows Enable Slot's slot type at 20:16, which the table makes RsvdZ.
layout erx_xhci_trb_noop_command wire_le 8 16
  padding 0 12
  field control u32 12
    bits cycle 0 0
    bits type 10 15
  meaning xhci_command_ring
  selects control.type xhci_trb_type.noop_command
  activates control.cycle

# Enable Slot (6.4.3.2, Table 6-59): the slot type at 20:16, 0, the type of a USB device's slot
# (Table 7-9 of the body text). It names no slot; its completion reports the one enabled.
layout erx_xhci_trb_enable_slot_command wire_le 8 16
  padding 0 12
  field control u32 12
    bits cycle 0 0
    bits type 10 15
    bits slot_type 16 20
  meaning xhci_command_ring
  selects control.type xhci_trb_type.enable_slot
  fixed control.slot_type 0
  activates control.cycle

# Disable Slot (6.4.3.3, Table 6-60): the slot at 31:24.
layout erx_xhci_trb_disable_slot_command wire_le 8 16
  padding 0 12
  field control u32 12
    bits cycle 0 0
    bits type 10 15
    bits slot 24 31
  meaning xhci_command_ring
  selects control.type xhci_trb_type.disable_slot
  derived control.slot
  activates control.cycle
  reach disable

# Address Device (6.4.3.4, Tables 6-61 and 6-62): the input context's address, 16-byte aligned,
# and the slot. It binds EP0's ring.
layout erx_xhci_trb_address_device_command wire_le 8 16

  padding 8 4
  field control u32 12
    bits cycle 0 0
    bits type 10 15
    bits slot 24 31
  meaning xhci_command_ring

# Set TR Dequeue Pointer (xhci.h:150-152; 6.4.3.9, Tables 6-67 to 6-69): the ring position,
# 16-byte aligned, with the cycle state the endpoint resumes with, both the sequencer's from the

layout erx_xhci_trb_set_dequeue_command wire_le 8 16

    bits dequeue_cycle 0 0
    bits pointer 4 63 shift 4 device_address
  padding 8 4
  field control u32 12
    bits cycle 0 0
    bits type 10 15
    bits endpoint 16 20
    bits slot 24 31
  meaning xhci_command_ring
  selects control.type xhci_trb_type.set_tr_dequeue
  derived parameter.dequeue_cycle
  derived parameter.pointer
  derived control.endpoint
  derived control.slot
  activates control.cycle

# A transfer event (xhci.h:233-277, xhci.c:951-960; the event-data flag from xHCI 1.2,
# 6.4.2.1, Table 6-39 of the body text): the TRB it reports on, the residue, the
# completion code, the slot and the device context index.
layout erx_xhci_trb_transfer_event wire_le 8 16
  field pointer u64 0 device_address
  field status u32 8
    bits residue 0 23
    bits code 24 31
  field control u32 12
    bits cycle 0 0
    bits event_data 2 2
    bits type 10 15
    bits endpoint 16 20
    bits slot 24 31
  meaning xhci_event_ring
  selects control.type xhci_trb_type.transfer_event
  activates control.cycle
  releases pointer xhci_transfer_ring when status.code xhci_completion.success xhci_completion.short_packet
  states status.residue
  states control.event_data
  states control.endpoint
  states control.slot

  states status.code
  states control.slot
